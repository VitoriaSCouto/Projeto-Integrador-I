model Vitima {
  id_vitima      Int      @id @default(autoincrement())
  nome           String
  cpf            String?  @default("Ainda não informado")
  telefone       String?  @unique @default("Ainda não informado")

  genero         String

model Deficiencia {
  id_deficiencia Int    @id @default(autoincrement())

model VitimaDeficiencia {
  id_vitima      Int
  id_deficiencia Int

  vitima      Vitima      @relation(fields: [id_vitima], references: [id_vitima])
  deficiencia Deficiencia @relation(fields: [id_deficiencia], references: [id_deficiencia])

  @@id([id_vitima, id_deficiencia])
}

enum OrigemRegistro {
  FAMILIAR // familiar q registrou
  EQUIPE_CAMPO // pode ser policia, SAR, qualquer orgao destinado a isso.. e cadastrou (sem familiar)
  ABRIGO // a pessoa ter aparecido num abrigo sem doc

  IML // corpo entrou no IML sem doc/id

enum Prioridade {
  BAIXA // 1 - sem urgencia imediaata
  MEDIA // 2 - monitoramento
  ALTA // 3 - busca ativa necessaria
  CRITICA // 4 - risco de vida (no caso seria alguem que precisa de medicamentos, crianças, idosos, PcDs... etc)
}

enum MetodoBusca {
  AEREA // equipe aerea - SAR; Helicoptero; Patrulha Area - reconhecimento de area.
  TERRESTRE
  AQUATICA // vitima possivelmente presa em enchentes 
  CAESFAREJADORES
  MERGULHO // vitima possivelmente levada a obito, tambem em enchentes.
  DRONE // mais comum doq parece
}

enum CriticidadeMedicamento {
  MONITORAR // vitima desaparecida pode ficar alguns dias sem 
  URGENTE // precisa até 24h, remedio controlado
  ESSENCIAL // risco de vida sem o medicamento
}

  status            StatusIML   @default(NAO_IDENTIFICADO)
  dataEntradaIML    DateTime
  dataIdentificacao DateTime?   // preenchida quando status → IDENTIFICADO

  observacoes       String?

  // quem do IML registrou / confirmou a identificação
  agenteId          Int
  agente            Agente @relation(fields: [agenteId], references: [id_agente])

  createdAt         DateTime @default(now()) @map("created_at")
